49SMGR_RESULT sensor_mgr_initialize(uint32_t number_of_sensors) {
50 SENSOR_REGISTER_RESULT result = sensor_register_initialize(number_of_sensors);
51 ESP_ERROR_CHECK(sensor_data_initialize(number_of_sensors));
52 switch (result) {
53 case SENSOR_CFG_OK:
54 ESP_LOGI(TAG, "Successfully initialized configurations for %i sensors",
55 number_of_sensors);
56 return SMGR_SUCCESS;
57 case SENSOR_CFG_LOW_MEMORY:
58 ESP_LOGE(TAG, "Could not initialize memory for sensor-configurations");
59 return SMGR_CATASTROPHIC_FAILURE;
60 default:
61 ESP_LOGE(TAG, "Unknown error occured");
62 return SMGR_UNKNOWN_ERROR;
63 };
64}

66SMGR_RESULT sensor_manager_read_sensor(uint32_t sensor_handle,
67 sensor_reading_T *reading) {
68 if (reading == NULL) {
69 ESP_LOGE(TAG, "Parameter 'reading' was not initialized");
70 return SMGR_UNKNOWN_ERROR;
71 }
72 sensor_T sensor;
73 if (sensor_register_fetch(sensor_handle, &sensor) != SENSOR_CFG_OK) {
74 ESP_LOGE(TAG, "Could not retrieve configuration for sensor %i",
75 sensor_handle);
76 abort();
77 }
78
79 uint32_t raw_value;
80 ESP_ERROR_CHECK(sensor_drv_read(sensor.sensor_gpio, &raw_value));
81
82 reading->sensor_id = sensor_handle;
83 reading->humidity_percentage =
84 get_humidity(raw_value, sensor.calibration_data);
85
86 ESP_LOGI(TAG, "Read humidity of %f percent from sensor %i",
87 reading->humidity_percentage, sensor_handle);
88 return SMGR_SUCCESS;
89}

92SMGR_RESULT sensor_mgr_add_sensor(uint32_t sensor_gpio,
93 sensor_handle_T *sensor_handle) {
94 uint32_t number_of_sensors = 0;
95 sensor_register_get_count(&number_of_sensors);
96 sensor_T new_sensor = {
97 .sensor_gpio = sensor_gpio,
98 .calibration_data = {.calibrated = false,
99 .air_measurement = 4095,
100 .water_measurement = 2717,
101 .air_reference = CONFIG_SENSOR_AIR_REFERENCE_VALUE,
102 .water_reference =
103 CONFIG_SENSOR_WATER_REFERENCE_VALUE}};
104
105 SENSOR_REGISTER_RESULT result =
106 sensor_register_insert(new_sensor, sensor_handle);
107
108 switch (result) {
109 case SENSOR_CFG_OK:
110 ESP_LOGI(TAG, "Successfully registered sensor [Handle: %i, GPIO: %i]",
111 *sensor_handle, sensor_gpio);
112 ESP_ERROR_CHECK(sensor_data_insert(0));
113 sensor_drv_initialize(sensor_gpio);
114
115 if (on_new_sensor != NULL) {
116 on_new_sensor(new_sensor);
117 }
118
119 return SMGR_SUCCESS;
120 case SENSOR_CFG_LOW_MEMORY:
121 ESP_LOGE(TAG, "Could not add sensor: Not enough memory");
122 return SMGR_CATASTROPHIC_FAILURE;
123 default:
124 ESP_LOGE(TAG,
125 "Could not register sensor with GPIO %i: Unknown error occured",
126 sensor_gpio);
127 return SMGR_UNKNOWN_ERROR;
128 }
129}

131static void sensor_polling_task(void *param) {
132 sensor_reading_T value_buffer;
133 uint32_t sensor_count = 0;
134 if (SENSOR_CFG_OK != sensor_register_get_count(&sensor_count)) {
135 ESP_LOGE(TAG, "Could not retrieve number of sensors");
136
137 vTaskDelete(NULL);
138 return;
139 }
140
141 if (sensor_count == 0) {
142 ESP_LOGW(TAG, "No sensors found. Have you forgotten to add them? Ending "
143 "polling-task");
144 vTaskDelete(NULL);
145 return;
146 }
147
148 while (1) {
149 ESP_LOGI(TAG, "Sensor polling started");
150
151 for (uint32_t i = 0; i < sensor_count; i++) {
152 ESP_LOGI(TAG, "Polling sensor %i/%i", i + 1, sensor_count);
153
154 if (sensor_manager_read_sensor(i, &value_buffer) != SMGR_SUCCESS) {
155 ESP_LOGE(TAG, "Error reading sensor %i", i);
156 continue;
157 }
158 sensor_data_update(value_buffer.sensor_id,
159 value_buffer.humidity_percentage);
160
161 if (on_new_measurement != NULL) {
162 on_new_measurement(value_buffer);
163 }
164 }
165
166 vTaskDelay(CONFIG_SENSOR_POLLING_TIME_IN_MS / portTICK_PERIOD_MS);
167 }
168 vTaskDelete(NULL);
169}

171SMGR_RESULT sensor_mgr_start_polling_task(void) {
172 BaseType_t result;
173 result =
174 xTaskCreatePinnedToCore(sensor_polling_task, "Sensor polling", 4 * 1024,
175 NULL, 4, NULL, CONFIG_SENSOR_POLLING_CORE);
176 if (result != pdPASS) {
177 ESP_LOGE(TAG, "Could not start sensor_polling-task");
178 return SENSOR_ERROR;
179 }
180
181 return SENSOR_SUCCESS;
182}
