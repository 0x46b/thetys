98
99 relais_handle_T relais_handle;

101 if (ESP_OK != relais_result) {
102 ESP_LOGE(TAG, "Could not add pump at GPIO %i", pump_gpio);
103 return ESP_FAIL;
104 }
105 new_plant.pump_handle = relais_handle;
106
107 ESP_ERROR_CHECK(plant_repo_insert(new_plant));
108 if (created_plant != NULL) {

110 }
