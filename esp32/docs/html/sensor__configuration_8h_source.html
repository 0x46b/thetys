28/*! Result-type to be able to determine the result of an operation */
29typedef enum SENSOR_CONFIGURATION_RESULT {
30 SENSOR_CFG_OK = 0, /*!< Everything was fine, operation successfull*/
31 SENSOR_CFG_ERROR =
32 1, /*!< Something went wrong, but we do not know what exactly */
33 SENSOR_CFG_LOW_MEMORY =
34 2, /*!< Not enough memory for completing the operation */
35 SENSOR_CFG_UNKNOWN_SENSOR_ID =
36 4, /*!< The operation was given a sensor-id that is not registered */
37 SENSOR_CFG_UNINITIALIZED = 5, /*!< Array that is given is not initialized */
38 SENSOR_CFG_INITIALIZATION_ERROR =
39 6, /*!< Dynamic array could not get initialized correctly */
40 SENSOR_CFG_PARAMETER_NULL = 7, /*!< Parameter is null */
41} SENSOR_CONFIGURATION_RESULT;

43/*! Calibration data used for calculating the humidity-percentage */
44typedef struct sensor_calibration_data_T {
45 bool calibrated; /*!< Determines, if the calibration was executed or if these
46 are default values */
47 uint32_t air_measurement; /*!< The value that gots measured by calibrate_air

49 uint32_t
50 water_measurement; /*!< The value that gots measure by calibrate_water if

52 uint32_t air_reference; /*!< Reference value, that we would expect for "no
53 water" */
54
55 uint32_t
56 water_reference; /*!< Reference value that we expect for "full water" */
57} sensor_calibration_data_T;

59/*! Configuration for a sensor */
60typedef struct sensor_T {
61 uint32_t sensor_gpio; /*!< The GPIO where the sensor is attached*/
62 sensor_calibration_data_T
63 calibration_data; /*!< Calibration data for the sensor */
64} sensor_configuration_T;

66/*! Dynamic list to be able to attach/remove sensors dynamically */
67typedef struct sensor_register_T {
68 sensor_configuration_T
69 *configurations; /*!< Dynamic array of all known sensor_configurations */
70 uint32_t size; /*!< Current size of the array (how much memory we have

72 uint32_t used; /*!< Number of used configurations */
73} sensor_configuration_list_T;

75/*! Initializes the dynamic-configuration array by allocating enough memory and
76 * setting initial values
77 * @param initialSize Initial size of the array
78 * @return SENSOR_CFG_OK The array got successfully
79 * initialized
80 * @return SENSOR_CFG_LOW_MEMORY Not enough memory to allocate
81 * enough space for initialSize-elements
82 */
83SENSOR_CONFIGURATION_RESULT
84sensor_register_initialize(size_t initialSize);
85
86/*! Inserts a configuration into the dynamic-array
87 * @param element The configuration that should get stored in the array
88 * @return SENSOR_CFG_OK The configuration got stored
89 * successfully
90 * @return SENSOR_CFG_LOW_MEMORY Not enough memory to allocate
91 * space needed
92 * @return SENSOR_CFG_UNINITIALIZED Array is not initialized,
93 * call init_configurations first
94 */
95SENSOR_CONFIGURATION_RESULT
96sensor_register_insert(sensor_configuration_T element, sensor_handle_T *handle);
97
98/*! Resets the dynamic array, frees all memory for the stored configurations
99 * @return SENSOR_CFG_OK The configuration got reset
100 * successfully
101 */
102SENSOR_CONFIGURATION_RESULT
103sensor_register_free();
104
105/*! Gets the configuration for a registered sensor with the id sensor_id
106 * @param[out] sensor_config pointer to an already allocated sensor_config
107 * @param[in] sensor_id Id of the sensor
108 * @returns SENSOR_CONFIG_OK Everything had worked
109 * @returns SENSOR_CFG_UNKNOWN_SENSOR_ID Sensor with the given id is not known
110 * (NOTE: sensor-ids start with 0)
111 */
112SENSOR_CONFIGURATION_RESULT
113sensor_register_fetch(sensor_configuration_T *sensor_config,
114 sensor_handle_T sensor_id);
