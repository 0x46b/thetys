27SENSOR_REGISTER_RESULT
28sensor_register_initialize(size_t initialSize) {
29 if (initialSize == 0) {
30 ESP_LOGE(TAG, "Initializing with 0 elements is not allowed.");
31 return SENSOR_CFG_INITIALIZATION_ERROR;
32 }
33
34 ESP_LOGI(TAG, "Initializing array for %i sensors.", initialSize);
35 sensor_configurations.configurations = malloc(initialSize * sizeof(sensor_T));
36
37 if (sensor_configurations.configurations == NULL) {
38 ESP_LOGE(TAG, "Malloc failed!");
39 return SENSOR_CFG_LOW_MEMORY;
40 }
41
42 sensor_configurations.used = 0;
43 sensor_configurations.size = initialSize;
44 initialized = true;
45
46 ESP_LOGI(TAG, "Successfully initialized memory for %i configurations.",
47 initialSize);
48 return SENSOR_CFG_OK;
49}

51SENSOR_REGISTER_RESULT
52sensor_register_insert(sensor_T element, sensor_handle_T *handle) {
53 if (!initialized) {
54 ESP_LOGE(TAG, "Try to insert into uninitialized array");
55 return SENSOR_CFG_UNINITIALIZED;
56 }
57
58 if (sensor_configurations.used == sensor_configurations.size) {
59 uint32_t new_size = sensor_configurations.size * 2;
60
61 ESP_LOGI(TAG,
62 "Not enough space for added sensor_config, resizing from %i to %i",
63 sensor_configurations.size, new_size);
64 sensor_T *new_array = realloc(sensor_configurations.configurations,
65 new_size * sizeof(sensor_T));
66 if (new_array == NULL) {
67 ESP_LOGE(TAG, "Resizing failed");
68 return SENSOR_CFG_LOW_MEMORY;
69 }
70 sensor_configurations.size = new_size;
71 sensor_configurations.configurations = new_array;
72 }
73
74 uint32_t new_index = sensor_configurations.used;
75 sensor_configurations.configurations[new_index] = element;
76 *handle = new_index;
77
78 ESP_LOGI(
79 TAG,
80 "Inserted new sensor-configuration [Handle: %i, GPIO %i](%i used total)",
81 *handle, element.sensor_gpio, sensor_configurations.used);
82
83 return SENSOR_CFG_OK;
84}

86SENSOR_REGISTER_RESULT
87sensor_register_free() {
88 free(sensor_configurations.configurations);
89 sensor_configurations.configurations = NULL;
90 sensor_configurations.used = 0;
91 sensor_configurations.size = 0;
92 initialized = false;
93
94 ESP_LOGI(TAG, "Reset configurations");
95 return SENSOR_CFG_OK;
96}

98SENSOR_REGISTER_RESULT
99sensor_register_fetch(sensor_handle_T sensor_handle, sensor_T *sensor_config) {
100 if (sensor_handle > sensor_configurations.used) {
101 ESP_LOGE(TAG, "Sensor with id %i is not known", sensor_handle);
102 return SENSOR_CFG_UNKNOWN_SENSOR_ID;
103 }
104 sensor_T stored_config =
105 sensor_configurations.configurations[(uint32_t)sensor_handle];
106 *sensor_config = stored_config;
107
108 ESP_LOGI(TAG, "Returning config for sensor %i (gpio:%i, air: %i, water:%i)",
109 sensor_handle, stored_config.sensor_gpio,
110 stored_config.calibration_data.air_measurement,
111 stored_config.calibration_data.water_measurement);
112 return SENSOR_CFG_OK;
113}
