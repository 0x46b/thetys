28/*! Result-type to be able to determine the result of an operation */
29typedef enum SENSOR_REGISTER_RESULT {
30 SENSOR_CFG_OK = 0, /*!< Everything was fine, operation successfull*/
31 SENSOR_CFG_ERROR =
32 1, /*!< Something went wrong, but we do not know what exactly */
33 SENSOR_CFG_LOW_MEMORY =
34 2, /*!< Not enough memory for completing the operation */
35 SENSOR_CFG_UNKNOWN_SENSOR_ID =
36 4, /*!< The operation was given a sensor-id that is not registered */
37 SENSOR_CFG_UNINITIALIZED = 5, /*!< Array that is given is not initialized */
38 SENSOR_CFG_INITIALIZATION_ERROR =
39 6, /*!< Dynamic array could not get initialized correctly */
40 SENSOR_CFG_PARAMETER_NULL = 7, /*!< Parameter is null */
41} SENSOR_REGISTER_RESULT;

67/*! Dynamic list to be able to attach/remove sensors dynamically */
68typedef struct sensor_register_T {
69 sensor_T
70 *configurations; /*!< Dynamic array of all known sensor_configurations */
71 uint32_t size; /*!< Current size of the array (how much memory we have
72 allocated) */
73 uint32_t used; /*!< Number of used configurations */
74} sensor_register_T;

76/*! Initializes the dynamic-configuration array by allocating enough memory and
77 * setting initial values
78 * @param initialSize Initial size of the array
79 * @return SENSOR_CFG_OK The array got successfully
80 * initialized
81 * @return SENSOR_CFG_LOW_MEMORY Not enough memory to allocate
82 * enough space for initialSize-elements
83 */
84SENSOR_REGISTER_RESULT
85sensor_register_initialize(size_t initialSize);
86
87/*! Inserts a configuration into the dynamic-array
88 * @param element The configuration that should get stored in the array
89 * @return SENSOR_CFG_OK The configuration got stored
90 * successfully
91 * @return SENSOR_CFG_LOW_MEMORY Not enough memory to allocate
92 * space needed
93 * @return SENSOR_CFG_UNINITIALIZED Array is not initialized,
94 * call init_configurations first
95 */
96SENSOR_REGISTER_RESULT
97sensor_register_insert(sensor_T element, sensor_handle_T *handle);

99/*! Resets the dynamic array, frees all memory for the stored configurations
100 * @return SENSOR_CFG_OK The configuration got reset
101 * successfully
102 */
103SENSOR_REGISTER_RESULT
104sensor_register_free();

106/*! Gets the configuration for a registered sensor with the id sensor_id
107 * @param[out] sensor_config pointer to an already allocated sensor_config
108 * @param[in] sensor_id Id of the sensor

110 * @returns SENSOR_CFG_UNKNOWN_SENSOR_ID Sensor with the given id is not known

113SENSOR_REGISTER_RESULT
