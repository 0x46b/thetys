12esp_err_t sensor_data_initialize(uint32_t initialSize) {
13 if (sensor_samples.samples != NULL) {
14 ESP_LOGW(TAG,
15 "Duplicate call to sensor_data_initialize(). Ignoring this one");
16 return ESP_ERR_NOT_ALLOWED;
17 }
18
19 ESP_LOGI(TAG, "Initializing array for %i samples", initialSize);
20 // If reset was called and the array gets reinitialized this could happen, so
21 // if samples is NULL but we already have a mutex, we can reuse it
22 if (sample_mutex == NULL) {
23 sample_mutex = xSemaphoreCreateMutex();
24 }
25
26 xSemaphoreTake(sample_mutex, portMAX_DELAY);
27 sensor_samples.samples = malloc(initialSize * sizeof(uint32_t));
28
29 if (sensor_samples.samples == NULL) {
30 ESP_LOGE(TAG, "Malloc failed!");
31 return ESP_ERR_NO_MEM;
32 }
33
34 sensor_samples.used = 0;
35 sensor_samples.size = initialSize;
36
37 xSemaphoreGive(sample_mutex);
38 ESP_LOGI(TAG, "Successfully initialized memory for %i samples.", initialSize);
39 return ESP_OK;
40}

42esp_err_t sensor_data_insert(uint32_t sample) {
43 xSemaphoreTake(sample_mutex, portMAX_DELAY);
44 if (sensor_samples.samples == NULL) {
45 ESP_LOGE(TAG, "Try to insert into uninitialized array");
46 return ESP_OK;
47 }
48
49 if (sensor_samples.used == sensor_samples.size) {
50 uint32_t new_size = sensor_samples.size * 2;
51
52 ESP_LOGI(TAG,
53 "Not enough space for adding new sample, resizing from %i to %i",
54 sensor_samples.size, new_size);
55 uint32_t *new_array =
56 realloc(sensor_samples.samples, sensor_samples.size * sizeof(uint32_t));
57 if (new_array == NULL) {
58 ESP_LOGE(TAG, "Resizing failed");
59 return ESP_ERR_NO_MEM;
60 }
61 sensor_samples.size = new_size;
62 sensor_samples.samples = new_array;
63 }
64
65 sensor_samples.samples[sensor_samples.used++] = sample;
66
67 xSemaphoreGive(sample_mutex);
68 ESP_LOGI(TAG, "Added sample '%i'. Now storing %i samples", sample,
69 sensor_samples.used);
70 return ESP_OK;
71}

73esp_err_t sensor_data_update(uint32_t id, uint32_t sample) {
74 xSemaphoreTake(sample_mutex, portMAX_DELAY);
75
76 if (id > sensor_samples.used) {
77 ESP_LOGE(TAG, "Try to insert with id '%i', but max-id is '%i'", id,
78 sensor_samples.used);
79 return ESP_ERR_NOT_SUPPORTED;
80 }
81 sensor_samples.samples[id] = sample;
82
83 xSemaphoreGive(sample_mutex);
84 ESP_LOGI(TAG, "Updated sample for sensor '%i' to '%i'", id, sample);
85 return ESP_OK;
86}
