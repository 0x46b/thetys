66/*! Takes an array of samples and cleans it from possible runaways
67 *
68 * It does it by calculating the average, the variance and the
69 * standard-deviation and filtering out every sample that deviates more than
70 * double the standard-deviation
71 * @param[in] sample_array Array with all samples we want to clean
72 * @param[in] number_of_samples Number of samples (or "size" of the given array)
73 * @param[out] cleaned_samples New array without runaway values
74 * @param[out] number_of_cleaned_samples Number of samples in the "clean" set
75 * @return #CALIB_SUCCESS Everything went fine
76 * @return #CALIB_LOWMEM Not enough memory to allocate for all
77 * samples
78 */
79CALIB_RESULT clean_samples(float *sample_array, uint32_t number_of_samples,
80 float *cleaned_samples,
81 uint32_t *number_of_cleaned_samples);
