28CALIB_RESULT clean_samples(float *sample_array, uint32_t number_of_samples,
29 float *cleaned_samples,
30 uint32_t *number_of_cleaned_samples) {
31 if (sample_array == NULL) {
32 ESP_LOGE(TAG, "sample_array is NULL");
33 return CALIB_UNINITIALIZED_PARAM;
34 }
35
36 if (number_of_cleaned_samples == NULL) {
37 ESP_LOGE(TAG, "number_of_cleaned_samples is NULL");
38 return CALIB_UNINITIALIZED_PARAM;
39 }
40
41 if (number_of_samples <= 0) {
42 // Do something, I dunno?
43 }
44
45 double variance = 0.0;
46 double standard_deviation = 0.0;
47 float average = 0.0;
48 CALIB_RESULT result =
49 get_sample_average(sample_array, number_of_samples, &average);
50
51 if (result != CALIB_SUCCESS) {
52 ESP_LOGE(TAG, "Something went very wrong here?");
53 return result;
54 }
55
56 for (int i = 0; i < number_of_samples; i++) {
57 variance += pow(sample_array[i] - average, 2);
58 }
59
60 standard_deviation = sqrt(variance / number_of_samples);
61
62 float *buffer = malloc(number_of_samples * sizeof(float));
63 if (buffer == NULL) {
64 ESP_LOGE(TAG, "Could not allocate enough memory for the return-values");
65 return CALIB_LOWMEM;
66 }
67
68 int target_index = 0;
69 for (int i = 0; i < number_of_samples; i++) {
70 double difference = fabs(sample_array[i] - average);
71
72 if (difference <= (2.0 * standard_deviation)) {
73 buffer[target_index] = sample_array[i];
74 target_index++;
75 }
76 }
77
78 *number_of_cleaned_samples = target_index + 1;
79 cleaned_samples = malloc(*number_of_cleaned_samples * sizeof(float));
80 memcpy(cleaned_samples, buffer, *number_of_cleaned_samples * sizeof(float));
81
82 free(buffer);
83 ESP_LOGI(TAG, "Removed %i samples (%i clean samples total)",
84 (number_of_samples - *number_of_cleaned_samples),
85 *number_of_cleaned_samples);
86 return CALIB_SUCCESS;
87}

89CALIB_RESULT get_sample_average(float *sample_array, uint32_t number_of_samples,
90 float *sample_average) {
91 if (sample_array == NULL) {
92 ESP_LOGE(TAG, "sample_array is NULL");
93 return CALIB_UNINITIALIZED_PARAM;
94 }
95
96 if (sample_average == NULL) {
97 ESP_LOGE(TAG, "sample_average is NULL");
98 return CALIB_UNINITIALIZED_PARAM;
99 }
100
101 double sum = 0.0;
102 for (int i = 0; i < number_of_samples; i++) {
103 sum += sample_array[i];
104 }
105
106 *sample_average = sum / number_of_samples;
107 return CALIB_SUCCESS;
108}

110CALIB_RESULT get_calibration_factor(sensor_T sensor_conf,
111 uint32_t number_of_samples,
112 float *calibration_factor) {
113 float sensor_reading_sum = 0;
114 uint32_t current_sensor_reading = 0;
115 float *sensor_samples = malloc(number_of_samples * sizeof(float));
116
117 if (sensor_samples == NULL) {
118 ESP_LOGE(TAG,
119 "Couldn't allocate memory for reading %i samples for calibration "
120 "of sensor %i",
121 number_of_samples, sensor_conf.sensor_gpio);
122 return CALIB_LOWMEM;
123 }
124
125 if (calibration_factor == NULL) {
126 ESP_LOGD(TAG, "Parameter calibration_factor is NULL");
127 return CALIB_UNINITIALIZED_PARAM;
128 }
129
130 SENSOR_RESULT result;
131
132 for (uint32_t sample_no = 0; sample_no < CONFIG_NUM_CALIBRATION_SAMPLES;
133 sample_no++) {
134 result = sensor_drv_read(sensor_conf.sensor_gpio, &current_sensor_reading);
135 if (result != SENSOR_SUCCESS) {
136 ESP_LOGE(TAG, "Error while reading sample");
137 abort();
138 }
139 sensor_samples[sample_no] = current_sensor_reading;
140 sensor_reading_sum += current_sensor_reading;
141 }
142
143 *calibration_factor = sensor_reading_sum / number_of_samples;
144
145 free(sensor_samples);
146 return CALIB_SUCCESS;
147}

149uint32_t get_humidity(int32_t raw_value,
150 sensor_calibration_data_T calibration) {
151 ESP_LOGI(TAG, "Calculating humidity-percentage (adc: %i, air: %i, water: %i)",
152 raw_value, calibration.air_measurement,
153 calibration.water_measurement);
154
155 float humidity_float =
156 (float)(raw_value - calibration.air_measurement) * 100.0f /
157 (float)(calibration.water_measurement - calibration.air_measurement);
158 float rounded_float = nearbyintf(humidity_float);
159
160 if (rounded_float < 0.0f) {
161 rounded_float = 0.0f;
162 } else if (rounded_float > 100.0f) {
163 rounded_float = 100.0f;
164 }
165
166 return (uint32_t)rounded_float;
167}
